```sql
-- A1 Illinois against the other 17 Big Ten schools, latest data year of each metric
-- Peer statistics leave Illinois out; the public median also leaves out Northwestern and USC.
-- Rank 1 is the highest value. Change runs from the most recent year at least five years before
-- the latest, which is six years for business_major_share_of_ug (IPEDS collects it in even years).
with metrics as (
    select
        *,
        case
            when metric in (
                'applicants', 'business_bachelors', 'business_masters',
                'business_bachelors_incl_307102', 'business_masters_incl_307102'
            ) then 'count'
            when metric = 'pell_gap_6yr' then 'gap'
            else 'rate'
        end as kind
    from "mart_peer_metrics"
    where is_illinois
),

latest as (
    select metric, max(data_year) as data_year
    from metrics
    group by metric
),

base_years as (
    select l.metric, max(m.data_year) as data_year
    from latest as l
    inner join metrics as m on m.metric = l.metric and m.data_year <= l.data_year - 5
    group by l.metric
),

paired as (
    select
        now.*,
        before.data_year_label as base_label,
        now.value - before.value as illinois_change,
        now.peer_median_excl_il - before.peer_median_excl_il as peer_change
    from metrics as now
    inner join latest as l on l.metric = now.metric and l.data_year = now.data_year
    left join base_years as b on b.metric = now.metric
    left join metrics as before on before.metric = b.metric and before.data_year = b.data_year
)

select
    metric as "Metric",
    data_year_label as "Data year",
    case kind
        when 'count' then printf('%,d', cast(round(value) as bigint))
        when 'gap' then printf('%+.1f pts', 100 * (value))
        else printf('%.1f%%', 100 * (value))
    end as "Illinois",
    case kind
        when 'count' then printf('%,d', cast(round(peer_median_excl_il) as bigint))
        when 'gap' then printf('%+.1f pts', 100 * (peer_median_excl_il))
        else printf('%.1f%%', 100 * (peer_median_excl_il))
    end as "Peer median",
    case kind
        when 'count' then printf('%,d', cast(round(peer_p25) as bigint)) || ' to ' || printf('%,d', cast(round(peer_p75) as bigint))
        when 'gap' then printf('%+.1f pts', 100 * (peer_p25)) || ' to ' || printf('%+.1f pts', 100 * (peer_p75))
        else printf('%.1f%%', 100 * (peer_p25)) || ' to ' || printf('%.1f%%', 100 * (peer_p75))
    end as "Peer IQR",
    case kind
        when 'count' then printf('%,d', cast(round(public_peer_median_excl_il) as bigint))
        when 'gap' then printf('%+.1f pts', 100 * (public_peer_median_excl_il))
        else printf('%.1f%%', 100 * (public_peer_median_excl_il))
    end as "Public peer median",
    rank_of_18 || ' of ' || n_ranked as "Rank",
    case kind
        when 'count' then printf('%+,d', cast(round(illinois_change) as bigint))
        else printf('%+.1f pts', 100 * (illinois_change))
    end as "Illinois change",
    case kind
        when 'count' then printf('%+,d', cast(round(peer_change) as bigint))
        else printf('%+.1f pts', 100 * (peer_change))
```

```sql
-- B1 Gies headcount by program family and modality, Fall 2017 to Fall 2026
-- DMI through Fall 2025, DAIR for Fall 2026.
with families as (
    select level, program_family, modality, fall_year, sum(headcount) as headcount
    from "mart_gies_enrollment"
    where fall_year between 2017 and 2026
    group by level, program_family, modality, fall_year
),

level_totals as (
    select level, 'All ' || level as program_family, 'all' as modality, fall_year,
        sum(headcount) as headcount
    from families
    group by level, fall_year
),

college_total as (
    select 'all' as level, 'All Gies' as program_family, 'all' as modality, fall_year,
        sum(headcount) as headcount
    from families
    group by fall_year
),

all_rows as (
    select * from families
    union all
    select * from level_totals
    union all
    select * from college_total
)

select
    level as "Level",
    program_family as "Program family",
    modality as "Modality",
    printf('%,d', cast(round(sum(headcount) filter (where fall_year = 2017)) as bigint)) as "Fall 2017",
    printf('%,d', cast(round(sum(headcount) filter (where fall_year = 2018)) as bigint)) as "Fall 2018",
```

```sql
-- C2 Fitted linear trend per year in undergraduate grades, covid_grading terms left out
-- Least-squares slope on term-level rates, fall and spring terms fitted separately.
with by_group as (
    select
        yearterm,
        year,
        term,
        is_gies,
        sum(students_graded) as graded,
        sum(grade_points) as points,
        sum(a_range) as a_range,
        sum(dfw) as dfw,
        sum(w) as w
    from "int_course_terms"
    where is_undergraduate and term in ('Fall', 'Spring')
    group by yearterm, year, term, is_gies
),

groups as (
    select yearterm, year, term, 'Gies' as courses, graded, points, a_range, dfw, w
    from by_group
    where is_gies
    union all
    select yearterm, year, term, 'Campus', sum(graded), sum(points), sum(a_range), sum(dfw), sum(w)
    from by_group
    group by yearterm, year, term
),

rates as (
    select
        courses,
        term,
        year,
        points / graded as gpa,
        a_range / graded as a_range_share,
        dfw / (graded + w) as dfw_rate
    from groups
    where yearterm not in (
        select yearterm from "stg_config__term_flags" where term_flag = 'covid_grading'
    )
)

select
    courses as "Courses",
    term as "Terms",
    count(*) || ' terms, ' || min(year) || ' to ' || max(year) as "Fitted on",
    printf('%+.3f', regr_slope(gpa, year)) as "GPA per year",
    printf('%+.2f pts', 100 * regr_slope(a_range_share, year)) as "A-range per year",
    printf('%+.2f pts', 100 * regr_slope(dfw_rate, year)) as "DFW per year",
    printf('%.2f', regr_r2(a_range_share, year)) as "R squared, A-range"
from rates
group by courses, term
order by courses desc, term
```

<details><summary>SQL</summary>

```sql
-- C3 Change in A-range share in Gies undergraduate courses, Fall 2015 to Fall 2025, split into within-course change, course mix, and their interaction
-- The split uses courses taught in both terms, weighted by their share of graded students.
-- Within-course: sum of w0 * (s1 - s0). Mix: sum of s0 * (w1 - w0). Interaction: sum of
-- (w1 - w0) * (s1 - s0). The three add up to the change among courses taught in both terms.
with courses as (
    select yearterm, subject, number, sum(students_graded) as graded, sum(a_range) as a_range
    from "int_course_terms"
    where is_gies and is_undergraduate and yearterm in ('2015-fa', '2025-fa')
    group by yearterm, subject, number
),

in_both as (
    select subject, number
    from courses
    group by subject, number
    having count(*) = 2
),

weighted as (
    select
        c.yearterm,
        c.subject,
        c.number,
        c.graded,
        c.graded / sum(c.graded) over (partition by c.yearterm) as weight,
        c.a_range / c.graded as share
    from courses as c
    inner join in_both as b on b.subject = c.subject and b.number = c.number
),

paired as (
    select
        f15.weight as w0,
        f25.weight as w1,
        f15.share as s0,
        f25.share as s1,
        f15.graded as graded_2015,
        f25.graded as graded_2025
    from weighted as f15
    inner join weighted as f25
        on f25.subject = f15.subject and f25.number = f15.number and f25.yearterm = '2025-fa'
    where f15.yearterm = '2015-fa'
),

parts as (
    select
        count(*) as n_courses,
        sum(graded_2015) as graded_2015,
        sum(graded_2025) as graded_2025,
        sum(w0 * s0) as share_2015,
        sum(w1 * s1) as share_2025,
        sum(w0 * (s1 - s0)) as within_course,
        sum(s0 * (w1 - w0)) as course_mix,
        sum((w1 - w0) * (s1 - s0)) as interaction
    from paired
),

all_courses as (
    select
        count(*) filter (where yearterm = '2015-fa') as n_2015,
        count(*) filter (where yearterm = '2025-fa') as n_2025,
        sum(graded) filter (where yearterm = '2015-fa') as graded_2015,
        sum(graded) filter (where yearterm = '2025-fa') as graded_2025,
        sum(a_range) filter (where yearterm = '2015-fa')
        / sum(graded) filter (where yearterm = '2015-fa') as share_2015,
        sum(a_range) filter (where yearterm = '2025-fa')
        / sum(graded) filter (where yearterm = '2025-fa') as share_2025
    from courses
)

select 1 as "Row", 'All courses: A-range share, Fall 2015' as "Measure",
    printf('%.1f%%', 100 * (share_2015)) || ' of ' || printf('%,d', cast(round(graded_2015) as bigint)) || ' graded in '
    || n_2015 || ' courses' as "Value"
from all_courses
union all
select 2, 'All courses: A-range share, Fall 2025',
    printf('%.1f%%', 100 * (share_2025)) || ' of ' || printf('%,d', cast(round(graded_2025) as bigint)) || ' graded in '
    || n_2025 || ' courses'
from all_courses
union all
select 3, 'All courses: change', printf('%+.1f pts', 100 * (share_2025 - share_2015)) from all_courses
union all
select 4, 'Courses taught in both terms: A-range share, Fall 2015',
    printf('%.1f%%', 100 * (share_2015)) || ' of ' || printf('%,d', cast(round(graded_2015) as bigint)) || ' graded in '
    || n_courses || ' courses'
from parts
union all
select 5, 'Courses taught in both terms: A-range share, Fall 2025',
    printf('%.1f%%', 100 * (share_2025)) || ' of ' || printf('%,d', cast(round(graded_2025) as bigint)) || ' graded'
from parts
union all
select 6, 'Courses taught in both terms: change', printf('%+.1f pts', 100 * (share_2025 - share_2015))
from parts
union all
select 7, 'Within-course change', printf('%+.1f pts', 100 * (within_course)) from parts
union all
select 8, 'Course mix', printf('%+.1f pts', 100 * (course_mix)) from parts
union all
select 9, 'Interaction', printf('%+.1f pts', 100 * (interaction)) from parts
union all
select 10, 'Courses entering or leaving (all-course change less the change in courses taught in both)',
    printf('%+.1f pts', 100 * ((a.share_2025 - a.share_2015) - (p.share_2025 - p.share_2015)))
from all_courses as a, parts as p
order by 1
```

```sql
-- C7 Online and in-person sections of the same Gies undergraduate course in the same long term, Fall 2018 on
-- Online and in-person sections often serve different students, so a gap here describes the
-- sections, not the format.
-- Schedule type is blank on every row from 2016-fa to 2018-sp.
with modes as (
    select
        yearterm,
        year,
        term,
        subject,
        number,
        sum(students_graded) filter (where mode = 'in_person') as ip_graded,
        sum(grade_points) filter (where mode = 'in_person') as ip_points,
        sum(a_range) filter (where mode = 'in_person') as ip_a_range,
        sum(dfw) filter (where mode = 'in_person') as ip_dfw,
        sum(w) filter (where mode = 'in_person') as ip_w,
        sum(students_graded) filter (where mode = 'online') as ol_graded,
        sum(grade_points) filter (where mode = 'online') as ol_points,
        sum(a_range) filter (where mode = 'online') as ol_a_range,
        sum(dfw) filter (where mode = 'online') as ol_dfw,
        sum(w) filter (where mode = 'online') as ol_w
    from "int_course_terms"
    where is_gies and is_undergraduate and term in ('Fall', 'Spring')
        and (year * 10 + case term when 'Spring' then 1 when 'Summer' then 2 when 'Fall' then 3 else 0 end) >= (2018 * 10 + case 'Fall' when 'Spring' then 1 when 'Summer' then 2 when 'Fall' then 3 else 0 end)
    group by yearterm, year, term, subject, number
)

select
    m.yearterm as "Term",
    coalesce(f.term_flag, '') as "Flag",
    m.subject || ' ' || m.number as "Course",
    printf('%,d', cast(round(ip_graded) as bigint)) as "In-person graded",
    printf('%,d', cast(round(ol_graded) as bigint)) as "Online graded",
    printf('%.2f', ip_points / ip_graded) as "In-person GPA",
    printf('%.2f', ol_points / ol_graded) as "Online GPA",
    printf('%.1f%%', 100 * (ip_a_range / ip_graded)) as "In-person A-range",
    printf('%.1f%%', 100 * (ol_a_range / ol_graded)) as "Online A-range",
    printf('%.1f%%', 100 * (ip_dfw / (ip_graded + ip_w))) as "In-person DFW",
    printf('%.1f%%', 100 * (ol_dfw / (ol_graded + ol_w))) as "Online DFW"
from modes as m
left join "stg_config__term_flags" as f on f.yearterm = m.yearterm
where ip_graded > 0 and ol_graded > 0
```

```sql
        sum(grade_points) filter (where mode = 'online')
        / sum(students_graded) filter (where mode = 'online') as ol_gpa,
        sum(a_range) filter (where mode = 'online')
        / sum(students_graded) filter (where mode = 'online') as ol_a_range,
        sum(dfw) filter (where mode = 'online')
        / sum(students_graded + w) filter (where mode = 'online') as ol_dfw
    from "int_course_terms"
    where is_gies and is_undergraduate and term in ('Fall', 'Spring')
        and (year * 10 + case term when 'Spring' then 1 when 'Summer' then 2 when 'Fall' then 3 else 0 end) >= (2018 * 10 + case 'Fall' when 'Spring' then 1 when 'Summer' then 2 when 'Fall' then 3 else 0 end)
    group by yearterm, year, term, subject, number
),

pairs as (
    select
        m.*,
        case when f.term_flag = 'covid_grading' then 'covid_grading terms' else 'Other terms' end
            as term_group
    from modes as m
    left join "stg_config__term_flags" as f on f.yearterm = m.yearterm
    where m.ip_gpa is not null and m.ol_gpa is not null
)

select
    term_group as "Terms",
    count(*) as "Course-terms",
    count(distinct subject || number) as "Courses",
    count(*) filter (where ol_a_range > ip_a_range) as "Online A-range higher",
    printf('%+.2f', median(ol_gpa - ip_gpa)) as "Median GPA difference",
    printf('%+.1f pts', 100 * (median(ol_a_range - ip_a_range))) as "Median A-range difference",
    printf('%+.1f pts', 100 * (median(ol_dfw - ip_dfw))) as "Median DFW difference"
from pairs
group by term_group
order by term_group desc
```

```sql
-- D3 Gies courses where primary sections and graded students moved in opposite directions, Fall 2019 to Fall 2025
-- Sections are distinct CRNs in the catalog; graded students come from the GPA dataset, which
-- leaves out sections with 20 or fewer graded students. Courses must appear in both terms in
-- both sources.
with sections as (
    select c.yearterm, c.subject, c.number, count(distinct c.crn) as sections
    from "stg_catalog" as c
    inner join "int_type_codes" as t on t.type_code = c.type_code
    where t.is_primary and c.yearterm in ('2019-fa', '2025-fa')
    group by c.yearterm, c.subject, c.number
),

students as (
    select yearterm, subject, number, sum(students_graded) as graded
    from "int_course_terms"
    where is_gies and yearterm in ('2019-fa', '2025-fa')
    group by yearterm, subject, number
),

courses as (
    select
        s.subject,
        s.number,
        max(s.sections) filter (where s.yearterm = '2019-fa') as sections_2019,
        max(s.sections) filter (where s.yearterm = '2025-fa') as sections_2025,
        max(g.graded) filter (where g.yearterm = '2019-fa') as graded_2019,
        max(g.graded) filter (where g.yearterm = '2025-fa') as graded_2025
    from sections as s
    inner join students as g
        on g.yearterm = s.yearterm and g.subject = s.subject and g.number = s.number
    group by s.subject, s.number
)

select
    case
        when sections_2025 < sections_2019 then 'Fewer sections, more students'
        else 'More sections, fewer students'
    end as "Direction",
    subject || ' ' || number as "Course",
    sections_2019 as "Sections, Fall 2019",
    sections_2025 as "Sections, Fall 2025",
    printf('%,d', cast(round(graded_2019) as bigint)) as "Graded, Fall 2019",
    printf('%,d', cast(round(graded_2025) as bigint)) as "Graded, Fall 2025"
from courses
where (sections_2025 < sections_2019 and graded_2025 > graded_2019)
    or (sections_2025 > sections_2019 and graded_2025 < graded_2019)
order by 1, subject, number
```

```sql
-- TRE lists by unit: Accountancy is ACCY, Business Administration BADM, Finance FIN, Business
-- BUS, Business Data and Innovation BDI, and the MBA units MBA. Business and Technical Writing
-- belongs to English and is left out. Unit names switch to mixed case in 2024-sp, so they are
-- matched without case. Counts only: names never leave data/raw. BUS has more TA listings than
-- primary sections in Falls 2017 to 2019, so its TA rate passes 100; it does against every
-- section type too, so the listings outnumber the sections rather than the denominator being
-- too narrow.
with listings as (
    select
        yearterm,
        case upper(unit)
            when 'ACCOUNTANCY' then 'ACCY'
            when 'BUSINESS ADMINISTRATION' then 'BADM'
            when 'FINANCE' then 'FIN'
            when 'BUSINESS' then 'BUS'
            when 'BUSINESS DATA AND INNOVATION' then 'BDI'
            when 'MBA' then 'MBA'
            when 'MBA PROGRAM' then 'MBA'
            when 'MASTERS OF BUSINESS ADMINISTRATION' then 'MBA'
        end as subject,
        role,
        listings
    from "stg_tre"
),

sections as (
    select c.yearterm, c.year, c.term, c.subject, count(distinct c.crn) as sections
    from "stg_catalog" as c
    inner join "int_type_codes" as t on t.type_code = c.type_code
    where t.is_primary and c.term in ('Fall', 'Spring')
    group by c.yearterm, c.year, c.term, c.subject
),

by_role as (
    select s.yearterm, s.year, s.term, s.subject, r.role, s.sections,
        coalesce(sum(l.listings), 0) as listings
    from sections as s
    cross join (values ('Instructor'), ('TA')) as r (role)
    left join listings as l
        on l.yearterm = s.yearterm and l.subject = s.subject and l.role = r.role
    where s.yearterm in (select distinct yearterm from "stg_tre")
    group by s.yearterm, s.year, s.term, s.subject, r.role, s.sections
)

select
    role as "Role",
    yearterm as "Term",
    printf(

        100.0 * sum(listings) filter (where subject = 'ACCY')
        / sum(sections) filter (where subject = 'ACCY')
    ) as "ACCY",
    printf(
        '%.1f',
        100.0 * sum(listings) filter (where subject = 'BADM')
        / sum(sections) filter (where subject = 'BADM')
    ) as "BADM",
    printf(

        100.0 * sum(listings) filter (where subject = 'BDI')
        / sum(sections) filter (where subject = 'BDI')
    ) as "BDI",
    printf(
        '%.1f',
        100.0 * sum(listings) filter (where subject = 'BUS')
        / sum(sections) filter (where subject = 'BUS')
    ) as "BUS",
    printf(
        '%.1f',
        100.0 * sum(listings) filter (where subject = 'FIN')
        / sum(sections) filter (where subject = 'FIN')
    ) as "FIN",
    printf(
        '%.1f',
        100.0 * sum(listings) filter (where subject = 'MBA')
        / sum(sections) filter (where subject = 'MBA')
    ) as "MBA",
    printf('%.1f', 100.0 * sum(listings) / sum(sections)) as "All Gies"
from by_role
group by role, yearterm, year, term
order by role, (year * 10 + case term when 'Spring' then 1 when 'Summer' then 2 when 'Fall' then 3 else 0 end)
```
